cmake_minimum_required(VERSION 3.21)
project(fltk-trayicon VERSION 0.1.0 LANGUAGES CXX)
option(FLTK_TRAYICON_BUILD_EXAMPLE "Build the FLTK tray demo" ${PROJECT_IS_TOP_LEVEL})
option(FLTK_TRAYICON_BUILD_TESTS "Build library contract tests" ${PROJECT_IS_TOP_LEVEL})
option(FLTK_TRAYICON_FETCH_FLTK "Fetch FLTK if no package is installed" ON)
if(NOT TARGET fltk::fltk)
    find_package(FLTK 1.4 CONFIG QUIET)
endif()
if(NOT TARGET fltk::fltk)
    if(NOT FLTK_TRAYICON_FETCH_FLTK)
        message(FATAL_ERROR "FLTK 1.4 is required. Set FLTK_DIR or enable FLTK_TRAYICON_FETCH_FLTK.")
    endif()
    include(FetchContent)
    foreach(feature TEST EXAMPLES FLUID FLTK_OPTIONS GL FORMS HTML_DOCS PDF_DOCS SHARED_LIBS)
        set(FLTK_BUILD_${feature} OFF CACHE BOOL "" FORCE)
    endforeach()
    FetchContent_Declare(fltk GIT_REPOSITORY https://github.com/fltk/fltk.git
        GIT_TAG 0d28470949323dedff75444d0011f0b5d38064f1)
    FetchContent_MakeAvailable(fltk)
endif()
add_library(fltk-trayicon STATIC fltk_trayicon.cpp)
add_library(fltk-trayicon::fltk-trayicon ALIAS fltk-trayicon)
target_compile_features(fltk-trayicon PUBLIC cxx_std_17)
target_include_directories(fltk-trayicon PUBLIC
    $<BUILD_INTERFACE:${CMAKE_CURRENT_SOURCE_DIR}>
    $<INSTALL_INTERFACE:include>)
target_link_libraries(fltk-trayicon PUBLIC fltk::images)
if(WIN32)
    target_sources(fltk-trayicon PRIVATE src/windows.cpp)
    target_compile_definitions(fltk-trayicon PRIVATE UNICODE _UNICODE WIN32_LEAN_AND_MEAN NOMINMAX)
    target_link_libraries(fltk-trayicon PRIVATE user32 shell32 gdi32)
elseif(APPLE)
    enable_language(OBJCXX)
    set_target_properties(fltk-trayicon PROPERTIES OBJCXX_STANDARD 17 OBJCXX_STANDARD_REQUIRED YES)
    target_sources(fltk-trayicon PRIVATE src/macos.mm)
    set_source_files_properties(src/macos.mm PROPERTIES COMPILE_OPTIONS "-fobjc-arc")
    target_link_libraries(fltk-trayicon PRIVATE "-framework AppKit")
elseif(CMAKE_SYSTEM_NAME STREQUAL "Linux")
    target_sources(fltk-trayicon PRIVATE src/linux.cpp)
    target_link_libraries(fltk-trayicon PRIVATE ${CMAKE_DL_LIBS})
else()
    message(FATAL_ERROR "Supported platforms: Windows, Linux and macOS.")
endif()
if(MSVC)
    target_compile_options(fltk-trayicon PRIVATE /W4 /utf-8)
else()
    target_compile_options(fltk-trayicon PRIVATE -Wall -Wextra -Wpedantic)
endif()
if(FLTK_TRAYICON_BUILD_EXAMPLE)
    add_executable(fltk-trayicon-demo WIN32 MACOSX_BUNDLE examples/demo.cpp)
    target_link_libraries(fltk-trayicon-demo PRIVATE fltk-trayicon::fltk-trayicon)
    if(MSVC)
        target_link_options(fltk-trayicon-demo PRIVATE /ENTRY:mainCRTStartup)
    endif()
endif()
if(FLTK_TRAYICON_BUILD_TESTS)
    enable_testing()
    add_executable(fltk-trayicon-tests tests/contracts.cpp)
    target_link_libraries(fltk-trayicon-tests PRIVATE fltk-trayicon::fltk-trayicon)
    add_test(NAME tray-contracts COMMAND fltk-trayicon-tests)
endif()
include(GNUInstallDirs)
include(CMakePackageConfigHelpers)
install(TARGETS fltk-trayicon EXPORT fltk-trayicon-targets ARCHIVE DESTINATION ${CMAKE_INSTALL_LIBDIR})
install(FILES fltk_trayicon.h fltk-trayicon_global.h DESTINATION ${CMAKE_INSTALL_INCLUDEDIR})
install(EXPORT fltk-trayicon-targets NAMESPACE fltk-trayicon:: DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-trayicon)
configure_package_config_file(cmake/fltk-trayicon-config.cmake.in
    "${CMAKE_CURRENT_BINARY_DIR}/fltk-trayicon-config.cmake"
    INSTALL_DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-trayicon)
write_basic_package_version_file("${CMAKE_CURRENT_BINARY_DIR}/fltk-trayicon-config-version.cmake" COMPATIBILITY SameMajorVersion)
install(FILES "${CMAKE_CURRENT_BINARY_DIR}/fltk-trayicon-config.cmake"
    "${CMAKE_CURRENT_BINARY_DIR}/fltk-trayicon-config-version.cmake" DESTINATION ${CMAKE_INSTALL_LIBDIR}/cmake/fltk-trayicon)
install(FILES LICENSE DESTINATION ${CMAKE_INSTALL_DATADIR}/licenses/fltk-trayicon)
